            border: 1px solid black;
        }

        /* 2. Bảng nâng cao sử dụng đầy đủ các thuộc tính CSS */
        .styled-table {
            /* Chiều rộng bảng */
            width: 100%;
            /* Gộp các đường viền thành 1 đường đơn */
            border-collapse: collapse;
            background-color: #ffffff;
            box-shadow: 0 4px 8px rgba(0, 0, 0, 0.05);
        }

        /* Định kiểu cho tiêu đề bảng (th) và ô dữ liệu (td) */
        .styled-table th, .styled-table td {
            /* Đường viền cho th và td */
            border: 1px solid #dddddd;
            /* Khoảng cách giữa đường viền và nội dung (padding) */
            padding: 12px 15px;
        }

        /* Chiều cao và căn chỉnh cho tiêu đề (th) */
        .styled-table th {
            height: 50px;
            /* Căn chỉnh theo chiều ngang (text-align) */
            text-align: left;
            /* Màu sắc: màu nền và màu chữ */
            background-color: #4CAF50;
            color: white;
        }

        /* Căn chỉnh theo chiều dọc (vertical-align) và màu sắc dòng */
        .styled-table td {
            /* Căn chỉnh văn bản ở giữa hoặc dưới (middle/bottom) */
            vertical-align: middle;
        }

        /* Xen kẽ màu nền cho các dòng (Zebra Striping) */
        .styled-table tbody tr:nth-child(even) {
            background-color: #f9f9f9;
        }

        /* Hiệu ứng rê chuột qua dòng */
        .styled-table tbody tr:hover {
            background-color: #f1f1f1;
        }
    </style>
